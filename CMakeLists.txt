cmake_minimum_required(VERSION 3.20)
project(huffblock VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)

set(HUFFBLOCK_SANITIZER "none" CACHE STRING "Sanitizer: none, address (ASan+UBSan), thread")
set_property(CACHE HUFFBLOCK_SANITIZER PROPERTY STRINGS none address thread)
option(HUFFBLOCK_PROFILE "Enable GNU gprof instrumentation" OFF)
option(HUFFBLOCK_BUILD_FUZZER "Build the Clang libFuzzer decoder target" OFF)
option(HUFFBLOCK_REFERENCE_CODEC "Use original per-bit paths for reproducible performance comparisons" OFF)
add_library(huffblock_build_options INTERFACE)
if(NOT HUFFBLOCK_SANITIZER MATCHES "^(none|address|thread)$")
    message(FATAL_ERROR "Unknown HUFFBLOCK_SANITIZER")
endif()

if(HUFFBLOCK_BUILD_FUZZER)
    if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang" OR WIN32 OR
       NOT HUFFBLOCK_SANITIZER STREQUAL "address")
        message(FATAL_ERROR "The fuzzer requires non-Windows Clang and HUFFBLOCK_SANITIZER=address")
    endif()
endif()
if(NOT HUFFBLOCK_SANITIZER STREQUAL "none")
    if(WIN32 OR NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang" OR HUFFBLOCK_PROFILE)
        message(FATAL_ERROR "Sanitizers require non-Windows GCC/Clang and cannot be combined with profiling")
    endif()
    if(HUFFBLOCK_SANITIZER STREQUAL "address")
        set(sanitize_flags -fsanitize=address,undefined -fno-omit-frame-pointer -fno-sanitize-recover=all)
    else()
        set(sanitize_flags -fsanitize=thread -fno-omit-frame-pointer)
    endif()
    target_compile_options(huffblock_build_options INTERFACE ${sanitize_flags})
    target_link_options(huffblock_build_options INTERFACE ${sanitize_flags})
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
        target_compile_options(huffblock_build_options INTERFACE -fno-pie)
        target_link_options(huffblock_build_options INTERFACE -no-pie)
    endif()
endif()
if(HUFFBLOCK_PROFILE)
    if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "GNU" OR WIN32)
        message(FATAL_ERROR "HUFFBLOCK_PROFILE requires non-Windows GCC")
    endif()
    target_compile_options(huffblock_build_options INTERFACE -pg)
    target_link_options(huffblock_build_options INTERFACE -pg)
endif()

set(huffblock_sources
    src/codec.cpp
    src/archive.cpp
    src/file_output.cpp
)
add_library(huffblock_lib ${huffblock_sources})
target_include_directories(huffblock_lib PUBLIC include)
target_compile_features(huffblock_lib PUBLIC cxx_std_20)
target_link_libraries(huffblock_lib PUBLIC Threads::Threads huffblock_build_options)
if(HUFFBLOCK_REFERENCE_CODEC)
    target_compile_definitions(huffblock_lib PRIVATE HUFFBLOCK_REFERENCE_CODEC=1)
endif()
if(HUFFBLOCK_BUILD_FUZZER)
    # Keep coverage instrumentation separate from normal CLI/test consumers.
    add_library(huffblock_fuzz_lib STATIC ${huffblock_sources})
    target_include_directories(huffblock_fuzz_lib PUBLIC include)
    target_compile_features(huffblock_fuzz_lib PUBLIC cxx_std_20)
    target_link_libraries(huffblock_fuzz_lib PUBLIC Threads::Threads huffblock_build_options)
    target_compile_options(huffblock_fuzz_lib PRIVATE -fsanitize=fuzzer-no-link)
    add_executable(huffblock_fuzz_decode tests/fuzz_decode.cpp)
    target_link_libraries(huffblock_fuzz_decode PRIVATE huffblock_fuzz_lib)
    target_compile_options(huffblock_fuzz_decode PRIVATE -fsanitize=fuzzer-no-link)
    target_link_options(huffblock_fuzz_decode PRIVATE -fsanitize=fuzzer)
endif()

add_executable(huffblock src/main.cpp)
target_link_libraries(huffblock PRIVATE huffblock_lib)
if(MINGW)
    target_link_options(huffblock PRIVATE -municode)
endif()

function(huffblock_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive-)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
    endif()
endfunction()
huffblock_warnings(huffblock_lib)
huffblock_warnings(huffblock)

include(CTest)
if(BUILD_TESTING)
    add_executable(huffblock_tests tests/tests.cpp)
    target_link_libraries(huffblock_tests PRIVATE huffblock_lib)
    huffblock_warnings(huffblock_tests)
    add_test(NAME codec_and_archive COMMAND huffblock_tests)
    add_executable(huffblock_parallel_tests tests/parallel.cpp)
    target_link_libraries(huffblock_parallel_tests PRIVATE huffblock_lib)
    target_include_directories(huffblock_parallel_tests PRIVATE src)
    huffblock_warnings(huffblock_parallel_tests)
    add_test(NAME parallel_and_publication COMMAND huffblock_parallel_tests)
    add_executable(huffblock_adversarial_tests tests/adversarial.cpp)
    target_link_libraries(huffblock_adversarial_tests PRIVATE huffblock_lib)
    target_include_directories(huffblock_adversarial_tests PRIVATE src)
    huffblock_warnings(huffblock_adversarial_tests)
    add_test(NAME malformed_and_limits COMMAND huffblock_adversarial_tests)
    add_test(NAME cli
        COMMAND ${CMAKE_COMMAND}
            -DHUFFBLOCK=$<TARGET_FILE:huffblock>
            -DTEST_DIR=${CMAKE_CURRENT_BINARY_DIR}/cli-tests
            -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli.cmake)
    set_tests_properties(codec_and_archive parallel_and_publication malformed_and_limits cli PROPERTIES TIMEOUT 120)
endif()
